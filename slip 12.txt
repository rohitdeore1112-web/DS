#include <stdio.h>
#define SIZE 5

int queue[SIZE], front = -1, rear = -1;

void Insert(int element) {
    if ((front == 0 && rear == SIZE - 1) || (rear == (front - 1) % (SIZE - 1))) return;
    else if (front == -1) front = rear = 0;
    else if (rear == SIZE - 1 && front != 0) rear = 0;
    else rear++;
    queue[rear] = element;
}

void Delete() {
    if (front == -1) return;
    if (front == rear) front = rear = -1;
    else if (front == SIZE - 1) front = 0;
    else front++;
}

void Display() {
    if (front == -1) return;
    if (rear >= front)
        for (int i = front; i <= rear; i++) printf("%d ", queue[i]);
    else {
        for (int i = front; i < SIZE; i++) printf("%d ", queue[i]);
        for (int i = 0; i <= rear; i++) printf("%d ", queue[i]);
    }
    printf("\n");
}

int main() {
    int choice, element;
    while (1) {
        scanf("%d", &choice);
        switch (choice) {
            case 1: scanf("%d", &element); Insert(element); break;
            case 2: Delete(); break;
            case 3: Display(); break;
            case 4: return 0;
        }
    }
}
