#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node* next;
};

struct Stack {
    struct Node* top;
};

struct Stack* createStack() {
    struct Stack* s = (struct Stack*)malloc(sizeof(struct Stack));
    s->top = NULL;
    return s;
}

void Push(struct Stack* s, int val) {
    struct Node* temp = (struct Node*)malloc(sizeof(struct Node));
    temp->data = val;
    temp->next = s->top;
    s->top = temp;
}

int Pop(struct Stack* s) {
    if (s->top == NULL) return -1; // Empty stack
    struct Node* temp = s->top;
    int val = temp->data;
    s->top = temp->next;
    free(temp);
    return val;
}

void Reverse(struct Stack* s) {
    struct Node* temp = s->top;
    while (temp != NULL) {
        printf("%d ", temp->data);
        temp = temp->next;
    }
    printf("\n");
}

int main() {
    struct Stack* s = createStack();
    int choice, val;
    while (1) {
        scanf("%d", &choice);
        switch (choice) {
            case 1:
                scanf("%d", &val);
                Push(s, val);
                break;
            case 2:
                val = Pop(s);
                if (val == -1) printf("Stack is empty\n");
                else printf("Popped: %d\n", val);
                break;
            case 3:
                Reverse(s);
                break;
            case 4:
                return 0;
        }
    }
}
