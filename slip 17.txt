//q1
#include <stdio.h>
#include <string.h>

struct Customer {
    int cust_id;
    char name[50];
    char address[100];
};

int main() {
    int n;
    scanf("%d", &n);
    struct Customer c[n];

    for (int i = 0; i < n; i++) {
        scanf("%d", &c[i].cust_id);
        scanf(" %[^\n]", c[i].name);
        scanf(" %[^\n]", c[i].address);
    }

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (c[j].cust_id > c[j + 1].cust_id) {
                struct Customer temp = c[j];
                c[j] = c[j + 1];
                c[j + 1] = temp;
            }
        }
    }

    for (int i = 0; i < n; i++) {
        printf("%d %s %s\n", c[i].cust_id, c[i].name, c[i].address);
    }

    return 0;
}

//q2.1
#include <stdio.h>
#include <string.h>

#define MAX 100

char stack[MAX];
int top = -1;

void push(char c) {
    stack[++top] = c;
}

char pop() {
    if (top == -1) return '\0';
    return stack[top--];
}

int isMatchingPair(char open, char close) {
    return (open == '(' && close == ')') ||
           (open == '{' && close == '}') ||
           (open == '[' && close == ']');
}

int main() {
    char str[MAX];
    fgets(str, MAX, stdin);
    str[strcspn(str, "\n")] = '\0';

    for (int i = 0; str[i] != '\0'; i++) {
        char c = str[i];
        if (c == '(' || c == '{' || c == '[') {
            push(c);
        } else if (c == ')' || c == '}' || c == ']') {
            if (top == -1 || !isMatchingPair(pop(), c)) {
                printf("Not Balanced\n");
                return 0;
            }
        }
    }

    if (top == -1) printf("Balanced\n");
    else printf("Not Balanced\n");

    return 0;
}
