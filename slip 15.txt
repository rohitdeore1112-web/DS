#include <stdio.h>
#include <string.h>

#define MAX 100

char stack[MAX];
int top = -1;

void push(char c) {
    if (top < MAX - 1) stack[++top] = c;
}

char pop() {
    if (top == -1) return '\0';
    return stack[top--];
}

int main() {
    char str[MAX], reversed[MAX];
    int i = 0;

    fgets(str, MAX, stdin);
    str[strcspn(str, "\n")] = '\0';

    while (str[i] != '\0') {
        push(str[i]);
        i++;
    }

    i = 0;
    char c;
    while ((c = pop()) != '\0') {
        reversed[i++] = c;
    }
    reversed[i] = '\0';

    printf("%s\n", reversed);
    return 0;
}
