#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *left, *right;
} Node;

Node* insert(Node* root, int val) {
    if (!root) {
        Node* newNode = (Node*)malloc(sizeof(Node));
        newNode->data = val;
        newNode->left = newNode->right = NULL;
        return newNode;
    }
    if (val < root->data)
        root->left = insert(root->left, val);
    else
        root->right = insert(root->right, val);
    return root;
}

void countNodes(Node* root, int *leaf, int *nonLeaf) {
    if (!root) return;
    if (!root->left && !root->right)
        (*leaf)++;
    else
        (*nonLeaf)++;
    countNodes(root->left, leaf, nonLeaf);
    countNodes(root->right, leaf, nonLeaf);
}

int main() {
    Node* root = NULL;
    int n, val, leaf = 0, nonLeaf = 0;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%d", &val);
        root = insert(root, val);
    }
    countNodes(root, &leaf, &nonLeaf);
    printf("Leaf nodes: %d\n", leaf);
    printf("Non-leaf nodes: %d\n", nonLeaf);
    return 0;
}
