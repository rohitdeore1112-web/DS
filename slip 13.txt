#include <stdio.h>
#define SIZE 5

int queue[SIZE], front = -1, rear = -1;

void Insert(int element) {
    if (rear == SIZE - 1) return;
    if (front == -1) front = 0;
    queue[++rear] = element;
}

void Delete() {
    if (front == -1 || front > rear) {
        front = rear = -1;
        return;
    }
    front++;
    if (front > rear) front = rear = -1;
}

int Length() {
    if (front == -1) return 0;
    return rear - front + 1;
}

int main() {
    int choice, element;
    while (1) {
        scanf("%d", &choice);
        switch (choice) {
            case 1: scanf("%d", &element); Insert(element); break;
            case 2: Delete(); break;
            case 3: printf("%d\n", Length()); break;
            case 4: return 0;
        }
    }
}
